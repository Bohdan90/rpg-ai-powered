{
 const string Root="/private/tmp/strategic-map-ux/";static double last;static string previous="";static VisualElement watched;static bool reloaded;

 public static void Launch(){EditorSceneManager.OpenScene("Assets/_Project/Scenes/TacticalGraybox.unity");EditorApplication.EnterPlaymode();}
 static bool Shown(VisualElement e){for(var v=e;v!=null;v=v.parent)if(v.resolvedStyle.display==DisplayStyle.None||v.resolvedStyle.visibility==Visibility.Hidden)return false;return true;}
 static void Tick(){if(Application.isBatchMode||!EditorApplication.isPlaying||EditorApplication.timeSinceStartup-last<.15)return;last=EditorApplication.timeSinceStartup;var p=UnityEngine.Object.FindAnyObjectByType<BattlePresenter>();VisualElement safeRoot=null;try{safeRoot=p?.HudRoot;}catch(NullReferenceException){}if(safeRoot==null){if(!reloaded){reloaded=true;EditorSceneManager.LoadSceneInPlayMode("Assets/_Project/Scenes/TacticalGraybox.unity",new UnityEngine.SceneManagement.LoadSceneParameters(UnityEngine.SceneManagement.LoadSceneMode.Single));}return;}
 var game=EditorWindow.GetWindow(Type.GetType("UnityEditor.GameView,UnityEditor"));
 if(watched!=p.HudRoot){watched=p.HudRoot;watched.RegisterCallback<PointerDownEvent>(e=>File.AppendAllText(Root+"results/pointers.txt",DateTime.UtcNow.ToString("O")+" "+e.position+" "+((VisualElement)e.target).name+"\n"),TrickleDown.TrickleDown);}
 if(File.Exists(Root+"command.txt")){string c=File.ReadAllText(Root+"command.txt").Trim();File.Delete(Root+"command.txt");
 if(c=="view"){game.maximized=true;game.Focus();}
 if(c.StartsWith("fixture:"))Fixture(p,c.Substring(8));
 if(c=="scale-contact"||c=="scale-portal") {p.StartProductionRoads();var fixture=p.Duel;var path=TravelScale08.Mainland.Path(31,41);typeof(DuelForce).GetProperty("Node").SetValue(fixture.West,c=="scale-contact"?path[2]:24);if(c=="scale-contact")typeof(DuelForce).GetProperty("Node").SetValue(fixture.East,path[3]);fixture.Seamless.Observe();p.DuelChanged();}
 if(c=="start08")p.StartProductionRoads();if(c=="start07")p.StartSeamless();
 if(c=="recreate"){EditorSceneManager.LoadSceneInPlayMode("Assets/_Project/Scenes/TacticalGraybox.unity",new UnityEngine.SceneManagement.LoadSceneParameters(UnityEngine.SceneManagement.LoadSceneMode.Single));return;}
 if(c.StartsWith("scroll:")){var e=p.HudRoot.Q(c.Substring(7));e?.GetFirstAncestorOfType<ScrollView>()?.ScrollTo(e);}
 if(c.StartsWith("focus:")){var a=c.Substring(6);p.SeamlessMap?.Focus(new WorldAddress(a[0]=='A'?WorldId.Frontier:WorldId.StoneValley,int.Parse(a.Substring(1))));}
 if(c.StartsWith("load:"))p.LoadDuel(c.Substring(5));
 }
 var w=p.Duel;var d=new Data{gx=game.position.x,gy=game.position.y,gw=game.position.width,rw=p.HudRoot.worldBound.width,rh=p.HudRoot.worldBound.height,persistentPath=Application.persistentDataPath,focus=EditorWindow.focusedWindow?.titleContent.text,world=w!=null&&w.CanSave?w.CaptureSave():null,hash=w!=null&&w.CanSave?w.CaptureSave().checksum:BattleStateHash.Compute(p.State),active=p.State.CurrentUnitId?.Value??0,records=p.Journal.Records.Count,preview=p.PreviewText,units=p.State.Units.Select(u=>new Unit{id=u.Id.Value,hp=u.Hp,armor=u.Armor,barrier=u.TemporaryBarrier,exhausted=u.IsExhausted,burn=u.BurnStacks,uses=u.CloseHealUsed,move=u.MovementRemaining,action=u.ActionAvailable,x=u.Position.X,y=u.Position.Y}).ToArray(),cells=Cells(p),buttons=p.HudRoot.Query<Button>().ToList().Where(b=>Shown(b)&&b.enabledInHierarchy&&b.worldBound.width>0).Select(b=>new Btn{name=b.name,text=b.text,x=b.worldBound.center.x,y=b.worldBound.center.y}).ToArray(),hud=string.Join("\n",p.HudRoot.Query<Label>().ToList().Where(Shown).Select(l=>l.text))};
 string json=JsonUtility.ToJson(d);File.WriteAllText(Root+"state.json",json);if(d.hash!=previous){previous=d.hash;File.AppendAllText(Root+"results/gui-trace.jsonl",json+"\n");}
 }
 static UnitState U(int id,UnitProfile p,Side side,int x,int y,int? hp=null,int? armor=null)=>new UnitState(new UnitId(id),side,p,new GridPosition(x,y),side==Side.West?Facing.East:Facing.West,hp:hp,armor:armor);
 static void Fixture(BattlePresenter p,string name){
 if(name=="initiative")p.ConfigureBattle(new[]{U(1,UnitProfile.FireMageTI,Side.West,5,8),U(2,UnitProfile.HumanHealerTI,Side.West,6,8),U(3,UnitProfile.HumanWarriorTI,Side.West,7,8),U(4,UnitProfile.HumanArcherTI,Side.East,18,8),U(5,UnitProfile.ElfWarriorTI,Side.East,19,8)},new Battlefield(23,17),42);
 else if(name=="heal")p.ConfigureBattle(new[]{U(1,UnitProfile.HumanHealerTI,Side.West,5,8,10),U(2,UnitProfile.HumanWarriorTI,Side.West,10,8,10),U(3,UnitProfile.HumanWarriorTI,Side.West,12,8,10),U(4,UnitProfile.HumanWarriorTI,Side.East,19,8)},new Battlefield(23,17),2);
 else if(name=="fire"||name=="ice")p.ConfigureBattle(new[]{U(1,name=="fire"?UnitProfile.FireMageTI:UnitProfile.IceMageTI,Side.West,5,8),U(2,UnitProfile.HumanWarriorTI,Side.West,6,8),U(3,UnitProfile.HumanWarriorTI,Side.East,8,8)},new Battlefield(23,17),2);
 else if(name=="barrier")p.ConfigureBattle(new[]{U(1,UnitProfile.FireMageTI,Side.West,5,8),U(2,UnitProfile.HumanWarriorTI,Side.West,6,8),U(3,UnitProfile.FireMageTI,Side.East,8,8)},new Battlefield(23,17),2);
 else if(name=="expiry")p.ConfigureBattle(new[]{U(1,UnitProfile.FireMageTI,Side.West,5,8),U(2,UnitProfile.HumanWarriorTI,Side.West,6,8),U(3,UnitProfile.HumanWarriorTI,Side.East,19,8)},new Battlefield(23,17),2);
 else p.ConfigureBattle(new[]{U(1,UnitProfile.HumanWarriorTI,Side.West,8,8),U(2,UnitProfile.HumanWarriorTI,Side.East,9,8,40,name=="spill"?4:16)},new Battlefield(23,17),name=="miss"?5u:name=="guard"?31u:1u);
 }
 static Btn[] Cells(BattlePresenter p){var surface=p.HudRoot.Q("board-input").worldBound;var camera=p.GetComponentInChildren<Camera>();return Enumerable.Range(0,23*17).Select(k=>{int x=k%23,y=k/23;var v=camera.WorldToViewportPoint(new Vector3(x,0,y));return new Btn{name=x+","+y,x=surface.x+v.x*surface.width,y=surface.y+(1-v.y)*surface.height};}).ToArray();}
 [Serializable]class Unit{public int id,hp,armor,barrier,burn,uses,move,x,y;public bool action,exhausted;}
 [Serializable]class Btn{public string name,text;public float x,y;}
 [Serializable]class Data{public float gx,gy,gw,rw,rh;public string persistentPath,focus,hash,hud;public int active,records;public string preview;public Unit[] units;public CrossroadsSaveData world;public Btn[] buttons,cells;}
}
